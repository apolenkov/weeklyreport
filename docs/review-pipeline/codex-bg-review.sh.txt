#!/bin/bash
# Background Codex review for the 4 Wave B changes that timed out under the
# foreground Bash 600s ceiling. Runs detached so each codex exec gets a real
# 1500s watchdog. Writes one item.completed JSON line per change to /tmp.
cd /Users/wrk/work/customreport || exit 1

review_one() {
  id="$1"
  pf=$(mktemp)
  cat > "$pf" <<EOF
You are an adversarial reviewer for the OpenSpec change openspec/changes/$id/ in the CustomReport monorepo. Your cwd IS the repo. You are STRICTLY READ-ONLY — change nothing.

Read the ENTIRE change directory openspec/changes/$id/ (proposal.md, specs/**/spec.md, design.md, tasks.md), then the product code it cites and any existing openspec/specs/** it touches.

Be skeptical and specific. Find concrete flaws: security holes the fix misses, race conditions, missing edge cases, requirements with no scenario coverage, scenarios that contradict the design, tasks that do not trace to a requirement, wrong assumptions about the actual code, conflicts with an existing durable spec, or a materially simpler correct approach. Respect repo conventions: hardcoded Russian is only a bug in apps/web JSX; ComingSoon is intentional; a single init migration is intentional (MVP); English-only prod UI is intentional.

Return ONLY a single minified JSON object, no prose, no markdown fence:
{"verdict":"APPROVED"|"REVISE","issues":[{"severity":"Blocker|High|Medium|Low","where":"file or artifact","problem":"...","fix":"one-line fix"}],"notes":"..."}
EOF
  out=$(mktemp)
  codex exec -s read-only --json "$(cat "$pf")" > "$out" 2>/dev/null &
  cp=$!
  ( sleep 1500; kill -9 $cp 2>/dev/null ) &
  wp=$!
  wait $cp 2>/dev/null
  kill -9 $wp 2>/dev/null
  grep '"type":"item.completed"' "$out" | tail -1 > "/tmp/wbcodex-$id.json"
  if [ -s "/tmp/wbcodex-$id.json" ]; then echo "OK   $id"; else echo "EMPTY $id"; fi
}

for id in scope-period-lifecycle-authz make-publish-atomic transactional-snapshot-restore period-uniqueness-db-constraint; do
  review_one "$id" &
done
wait
echo "ALL-DONE"
