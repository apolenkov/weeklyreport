export const meta = {
  name: "codex-review-wave-cd",
  description:
    "Adversarial Codex review (read-only) of each Wave C+D OpenSpec change. One Codex pass per change, 590s watchdog. Returns VERDICT + issues. No edits.",
  phases: [{ title: "Review", detail: "one Codex reviewer per change (codex exec, read-only)" }],
};

const CHANGES = [
  "fix-analytics-date-matching",
  "fix-reminders-escalation-dedupe",
  "harden-web-vitals-endpoint",
  "protect-metrics-endpoint",
  "guard-metrics-scrape-db-failure",
  "fix-period-datetime-tz",
  "fix-risk-stage-integrity-impl",
  "key-charts-by-id",
  "fix-dashboard-nan-pending",
  "fix-heap-health-threshold",
  "enforce-smtp-config",
  "serialize-autosave",
  "observe-prisma-pool",
  "ci-deploy-supplychain-hardening",
  "observability-and-perf-hardening",
  "code-health-and-test-cleanup",
  "input-and-error-hardening",
];

const SCHEMA = {
  type: "object",
  required: ["changeId", "available", "verdict", "issues"],
  properties: {
    changeId: { type: "string" },
    available: { type: "boolean" },
    verdict: { enum: ["APPROVED", "REVISE", "UNKNOWN"] },
    issues: {
      type: "array",
      items: {
        type: "object",
        required: ["severity", "where", "problem", "fix"],
        properties: {
          severity: { enum: ["Blocker", "High", "Medium", "Low"] },
          where: { type: "string" },
          problem: { type: "string" },
          fix: { type: "string" },
        },
      },
    },
    notes: { type: "string" },
  },
};

const reviewPrompt = (id) =>
  `You are an adversarial reviewer for an OpenSpec change proposal in the CustomReport monorepo. Your cwd IS the repo. You are STRICTLY READ-ONLY — change nothing.

Read the ENTIRE change directory openspec/changes/${id}/ : proposal.md, every specs/**/spec.md delta, design.md, tasks.md. Then read the product code it cites and any existing openspec/specs/** it touches.

WORK EFFICIENTLY — spot-check the cited files, do NOT crawl the whole repo. Emit your JSON before you run long.

Be skeptical and specific. Find concrete flaws: security holes, race conditions, missing edge cases, requirements with no scenario coverage, scenarios that contradict the design, tasks that do not trace to a requirement, wrong assumptions about the actual code, conflicts with an existing durable spec, internal contradictions, or a materially simpler approach. Respect repo conventions: hardcoded Russian only a bug in apps/web JSX; ComingSoon intentional; single init migration intentional (MVP); English-only prod UI intentional; markdownlint removed (never flag pnpm lint:md).

Return ONLY a single minified JSON object, no prose, no markdown fence:
{"verdict":"APPROVED"|"REVISE","issues":[{"severity":"Blocker|High|Medium|Low","where":"file or artifact","problem":"...","fix":"one-line fix"}],"notes":"..."}
Use APPROVED if sound enough to implement; REVISE only if a material problem remains.`;

phase("Review");
const results = await parallel(
  CHANGES.map((id) => () => {
    const bash = `F=$(mktemp "\${TMPDIR:-/tmp}/codexcd-${id}.XXXXXX")
cat > "$F" <<'CODEXCD_EOF'
${reviewPrompt(id)}
CODEXCD_EOF
O=$(mktemp)
codex exec -s read-only --json "$(cat "$F")" > "$O" 2>/dev/null &
P=$!; ( sleep 590; kill -9 $P 2>/dev/null ) & W=$!
wait $P 2>/dev/null; kill -9 $W 2>/dev/null
grep '"type":"item.completed"' "$O" | tail -1`;
    return agent(
      `You are a RELAY for OpenAI Codex reviewing OpenSpec change "${id}". Run Codex
read-only and return ITS verdict verbatim — do NOT review it yourself.

Run this EXACT Bash as a SINGLE invocation:

${bash}

The printed line is a JSON item.completed whose .item.text holds Codex's answer
object {"verdict":...,"issues":[...],"notes":...}. Parse it and return as your
structured output with changeId:"${id}" and available:true. If empty / timed out /
unparseable, return changeId:"${id}", available:false, verdict:"UNKNOWN",
issues:[], notes explaining what went wrong.`,
      {
        label: `codex-review:${id}`,
        phase: "Review",
        model: "sonnet",
        effort: "medium",
        schema: SCHEMA,
      },
    );
  }),
);

return { reviews: results.filter(Boolean) };
