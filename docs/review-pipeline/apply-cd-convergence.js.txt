export const meta = {
  name: "apply-cd-convergence",
  description:
    "Apply round-2 convergence Codex findings to the 12 REVISE Wave C/D changes. Verify each finding as a candidate against artifact + code, apply valid ones, reject spurious with a reason, re-validate --strict. No product code.",
  phases: [{ title: "Revise", detail: "one reviser per REVISE change" }],
};

const REVIEW_FILE = "docs/repo-excellence-review-2026-06-28.wave-cd.convergence-review.json";
const CHANGES = [
  "fix-reminders-escalation-dedupe", "harden-web-vitals-endpoint", "protect-metrics-endpoint",
  "fix-risk-stage-integrity-impl", "fix-dashboard-nan-pending", "fix-heap-health-threshold",
  "serialize-autosave", "observe-prisma-pool", "ci-deploy-supplychain-hardening",
  "observability-and-perf-hardening", "code-health-and-test-cleanup", "input-and-error-hardening",
];

const SCHEMA = {
  type: "object",
  required: ["changeId", "validated", "applied", "rejected"],
  properties: {
    changeId: { type: "string" }, validated: { type: "boolean" },
    applied: { type: "array", items: { type: "object", required: ["issue", "how"], properties: { issue: { type: "string" }, how: { type: "string" } } } },
    rejected: { type: "array", items: { type: "object", required: ["issue", "why"], properties: { issue: { type: "string" }, why: { type: "string" } } } },
    notes: { type: "string" },
  },
};

const GUARD = `Treat each finding as a CANDIDATE — open the artifact + cited code (incl. generated
Prisma types), CONFIRM real and not already handled, THEN apply; reject with reason otherwise.
Sweep ALL artifacts so wording agrees. A cross-change dependency finding (e.g. depends on
set-express-trust-proxy) is resolved by a [Dependency] note in design.md, not by re-specifying
the other change. Repo conventions: Russian only a bug in apps/web JSX; ComingSoon intentional;
single init migration intentional (MVP); English-only prod UI intentional; never tell impl to
run pnpm lint:md. Edit OpenSpec ARTIFACTS only — no product code, no tasks marked done.`;

phase("Revise");
const results = await parallel(
  CHANGES.map((id) => () =>
    agent(
      `Revise ONE OpenSpec change by applying valid round-2 convergence findings. Edit only
openspec/changes/${id}/ artifacts.
STEP 1: Read ${REVIEW_FILE}; find the object changeId === "${id}"; its "issues" are the findings.
STEP 2: For each, open the cited artifact + product code; if REAL and not already handled, apply
the fix to the right artifact and sweep siblings for stale wording. Delta rules: "## ADDED/MODIFIED
Requirements", "### Requirement:" SHALL on first body line, each has a "#### Scenario:" GIVEN/WHEN/THEN.
If wrong/already-covered, record in "rejected".
STEP 3: \`openspec validate "${id}" --strict\` until exit 0.
Return {changeId, validated, applied, rejected, notes}.
${GUARD}`,
      { label: `cdconv:${id}`, phase: "Revise", model: "opus", effort: "high", schema: SCHEMA },
    ),
  ),
);

const done = results.filter(Boolean);
log(`Applied: ${done.reduce((n,r)=>n+(r.applied?.length||0),0)}, rejected ${done.reduce((n,r)=>n+(r.rejected?.length||0),0)}, validated ${done.filter(r=>r.validated).length}/${done.length}`);
return { revised: done };
