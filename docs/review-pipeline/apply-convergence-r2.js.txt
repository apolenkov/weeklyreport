export const meta = {
  name: "apply-convergence-r2",
  description:
    "Apply the round-2 convergence Codex findings to the 13 REVISE changes. One agent per change: verify each finding as a candidate against the artifact + code, apply valid ones, reject spurious with a note, re-validate --strict. No product code changes.",
  phases: [{ title: "Revise", detail: "one reviser per REVISE change, verify-then-apply, validate --strict" }],
};

const REVIEW_FILE = "docs/repo-excellence-review-2026-06-28.convergence-review.json";

const CHANGES = [
  "fix-prod-web-build-args",
  "route-pdfs-and-authz",
  "scope-analytics-to-group",
  "scope-leadform-revoke-authz",
  "add-section-optimistic-lock",
  "guard-autopublish-archived",
  "scope-period-lifecycle-authz",
  "scope-reminder-authz",
  "make-publish-atomic",
  "transactional-snapshot-restore",
  "period-uniqueness-db-constraint",
  "scope-leadform-getbyperiod",
  "map-domain-errors",
];

const SCHEMA = {
  type: "object",
  required: ["changeId", "validated", "applied", "rejected"],
  properties: {
    changeId: { type: "string" },
    validated: { type: "boolean" },
    applied: {
      type: "array",
      items: { type: "object", required: ["issue", "how"], properties: { issue: { type: "string" }, how: { type: "string" } } },
    },
    rejected: {
      type: "array",
      items: { type: "object", required: ["issue", "why"], properties: { issue: { type: "string" }, why: { type: "string" } } },
    },
    notes: { type: "string" },
  },
};

const GUARD = `
Treat each finding as a CANDIDATE — open the actual artifact + cited code, CONFIRM
it is real and not already addressed, THEN apply; reject with reason otherwise. Many
round-2 findings are INTERNAL DESYNC: the first revision updated design/spec/tasks
but left proposal.md (or a goal line, or the capability summary) stating the OLD
approach. Fixing those means making ALL artifacts state the SAME single approach —
check proposal.md, design goals, capability summary, and out-of-scope scenarios for
stale wording, not just the spot the finding names. A finding may also correct a
false premise (e.g. a Prisma constraint that does not actually exist) — verify
against the GENERATED Prisma types / real code before rewriting around it. Respect
repo conventions: hardcoded Russian only a bug in apps/web JSX; ComingSoon
intentional; single init migration intentional (MVP); English-only prod UI
intentional; markdownlint removed (never tell implementers to run pnpm lint:md). You
edit OpenSpec change ARTIFACTS only — no product code, do NOT mark tasks done.`;

phase("Revise");
const results = await parallel(
  CHANGES.map((id) => () =>
    agent(
      `Revise ONE OpenSpec change by applying the valid round-2 convergence findings.
You edit only openspec/changes/${id}/ artifacts.

CHANGE: ${id}

STEP 1: Read ${REVIEW_FILE} (a JSON array). Find the object whose changeId === "${id}";
its "issues" array holds the round-2 Codex findings.

STEP 2: For each finding, open the cited artifact in openspec/changes/${id}/ and the
cited product code (incl. generated Prisma types when a DB/type claim is made). If
REAL and not already handled, apply the fix to the right artifact AND sweep the other
artifacts (proposal/design/spec/tasks) for the same stale wording so they all agree.
Keep delta-spec rules: "## ADDED/MODIFIED Requirements", "### Requirement:" SHALL on
first body line, every requirement has a "#### Scenario:" GIVEN/WHEN/THEN. If a
finding is wrong or already covered, record it in "rejected" with the reason.

STEP 3: Run \`openspec validate "${id}" --strict\` and fix until it exits 0.

Return {changeId, validated, applied:[{issue,how}], rejected:[{issue,why}], notes}.
${GUARD}`,
      { label: `revise2:${id}`, phase: "Revise", model: "opus", effort: "high", schema: SCHEMA },
    ),
  ),
);

const done = results.filter(Boolean);
log(
  `Round-2 revised ${done.length}/${CHANGES.length}; validated ${done.filter((r) => r.validated).length}; ` +
    `applied ${done.reduce((n, r) => n + (r.applied?.length || 0), 0)}, ` +
    `rejected ${done.reduce((n, r) => n + (r.rejected?.length || 0), 0)}.`,
);
return { revised: done };
