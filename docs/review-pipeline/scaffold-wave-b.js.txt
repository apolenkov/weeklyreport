export const meta = {
  name: "scaffold-wave-b",
  description:
    "Scaffold Wave B P1 OpenSpec changes (authz + integrity) from the excellence-review backlog (proposal/specs/design/tasks) and validate each. Authors OpenSpec change artifacts only — NO code implementation.",
  phases: [{ title: "Scaffold", detail: "one agent per P1 change, validate --strict" }],
};

const CHANGES = [
  { changeId: "scope-period-lifecycle-authz", model: "opus", dedup: "period-dual-state, period-lifecycle-ui", scope: "api/period: publish/reopen target derived from the scoped active period (not an arbitrary client reportId); reject archived; arbitrary reportId is admin-only (council: Codex)" },
  { changeId: "scope-reminder-authz", model: "opus", dedup: "repository-scope-guard, scope-accessed-audit, admin-reminders-ui", scope: "api/reminders: thread ctx.session into dispatch/list/execute; for non-admin head apply groupScopeFilter on ruleId/reportId/leadId (council: Codex)" },
  { changeId: "enforce-deactivation-pdf-bullboard", model: "opus", dedup: "", scope: "api/auth: extract the isRevoked+isActive check into one shared helper and call it in BOTH buildPdfExportSession (PDF export) and the Bull Board guard (currently both skip it)" },
  { changeId: "make-publish-atomic", model: "opus", dedup: "manual-publish-telemetry, period-dual-state", scope: "api/publication + period: replace check-then-write with a compare-and-set updateMany where status!=published; increment the publish metric only when count===1; resolve the target period from DB to avoid a stale activeReportId" },
  { changeId: "transactional-snapshot-restore", model: "opus", dedup: "", scope: "api/admin: wrap the saveSnapshot phases in a single $transaction; replace the full destructive reconcile with targeted upsert/delete by id so a mid-restore failure cannot leave the report half-wiped" },
  { changeId: "period-uniqueness-db-constraint", model: "opus", dedup: "period-dual-state, schema-conventions", scope: "prisma + api/admin: add the partial unique index the comment promises (or remove the stale comment) and catch P2002 in ensureWeeklyReport instead of an app-level TOCTOU check" },
  { changeId: "scope-leadform-getbyperiod", model: "opus", dedup: "repository-scope-guard, scope-accessed-audit", scope: "api/report-sections: getByPeriod returns a co-lead's whole foreign section over one shared product — restrict the foreign section to shared products only; add a contract test for the exclusion" },
  { changeId: "map-domain-errors", model: "sonnet", dedup: "client-error-policy", scope: "api/error-mapping: throw typed domain errors from lead-form.service.ts / admin-*.queries.ts and add CONFLICT/NOT_FOUND/FORBIDDEN branches in mapDomainError so domain failures stop surfacing as 500 'Unexpected error'" },
];
if (CHANGES.length === 0) {
  log("No changes to scaffold — aborting.");
  return { error: "no-changes" };
}

const GUARD = `
REPO CONVENTIONS (do not contradict): hardcoded Russian is only a bug inside
apps/web JSX; ComingSoon is intentional deferral; a single init migration kept in
sync with the schema is intentional (MVP disposable DB — reshape schema.prisma +
hand-edit the init migration, do NOT author an incremental alter-table migration);
English-only prod UI is a decision. Use the canonical terms from CONTEXT.md. This
is REVIEW->PLAN: you author the OpenSpec change (proposal/specs/design/tasks) but
DO NOT modify any product code and DO NOT mark tasks done.`;

const SCHEMA = {
  type: "object",
  required: ["changeId", "created", "validated", "mode"],
  properties: {
    changeId: { type: "string" },
    created: { type: "boolean" },
    validated: { type: "boolean" },
    capability: { type: "string" },
    mode: { enum: ["new-capability", "modify-existing", "mixed", "failed"] },
    notes: { type: "string" },
  },
};

phase("Scaffold");
const results = await parallel(
  CHANGES.map((c) => () =>
    agent(
      `Author ONE OpenSpec change for the CustomReport repo (OpenSpec CLI 1.4.1,
spec-driven schema). You author the change ARTIFACTS only — do NOT implement any
product code, do NOT check off tasks as done.

CHANGE NAME: ${c.changeId}
SCOPE (one-line): ${c.scope}
DEDUP TARGET specs to check FIRST: ${c.dedup || "(none suggested — run `openspec list --specs` and look for an overlapping capability)"}

STEP 1 — get the finding detail (do not invent it):
  Read docs/repo-excellence-review-2026-06-28.data.json and find the backlog
  item(s) whose changeId/recommendation matches "${c.changeId}". Read the matching
  section of docs/repo-excellence-review-2026-06-28.md for context. Extract: the
  affected files, the actual problem, the recommended fix, the blast radius.

STEP 2 — decide capability + mode (content, not name match):
  Read the dedup-target spec body at openspec/specs/<target>/spec.md if any was
  suggested, and skim \`openspec list --specs\`. If an existing capability already
  governs this behavior, author a MODIFIED/ADDED delta against THAT capability
  name (mode "modify-existing"). Otherwise create a new kebab-case capability
  (mode "new-capability").

STEP 3 — scaffold:
  If openspec/changes/${c.changeId}/ does NOT already exist, run:
    openspec new change "${c.changeId}"

STEP 4 — author artifacts in dependency order:
  Run \`openspec status --change "${c.changeId}" --json\` for the build order and
  per-artifact outputPath. For EACH artifact (proposal -> specs -> design -> tasks):
    - \`openspec instructions <artifact-id> --change "${c.changeId}" --json\`
    - read every dependency file it names, then write the artifact to its
      outputPath using the template as skeleton and the instruction as a CONSTRAINT
      (the instruction/template text must NOT appear in the output file).
  Delta spec rules: file under specs/<capability>/spec.md; headers exactly
  "## ADDED Requirements" / "## MODIFIED Requirements"; each "### Requirement:" is
  phrased "The system SHALL ..." with the normative SHALL clause on the FIRST body
  line; every requirement has >=1 "#### Scenario:" with GIVEN / WHEN / THEN.
  tasks.md is an implementation PLAN (unchecked checkboxes), not completed work.
  Keep proposal Why/What-Changes/Capabilities/Impact tight and specific.

STEP 5 — validate:
  Run \`openspec validate "${c.changeId}" --strict\` and fix the artifacts until it
  exits 0. If it says "Unknown item", add \`--type change\`.

Return {changeId, created, validated, capability, mode, notes}. In notes, record
the capability name, whether you extended or created it, and any validation issue
you could not resolve.
${GUARD}`,
      {
        label: `scaffold:${c.changeId}`,
        phase: "Scaffold",
        model: c.model || "sonnet",
        effort: "high",
        schema: SCHEMA,
      },
    ),
  ),
);

const done = results.filter(Boolean);
log(
  `Scaffolded ${done.filter((r) => r.created).length}/${CHANGES.length}; ` +
    `validated ${done.filter((r) => r.validated).length}.`,
);
return { scaffolded: done };
