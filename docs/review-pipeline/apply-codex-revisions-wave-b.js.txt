export const meta = {
  name: "apply-codex-revisions-wave-b",
  description:
    "Apply the valid Codex review findings to each Wave B OpenSpec change. One agent per change: verify each finding as a candidate against the artifact + code, apply valid ones, reject spurious with a note, re-validate --strict. No product code changes.",
  phases: [{ title: "Revise", detail: "one reviser per change, verify-then-apply, validate --strict" }],
};

const REVIEW_FILE = "docs/repo-excellence-review-2026-06-28.wave-b.codex-review.json";

const CHANGES = [
  "scope-period-lifecycle-authz",
  "scope-reminder-authz",
  "enforce-deactivation-pdf-bullboard",
  "make-publish-atomic",
  "transactional-snapshot-restore",
  "period-uniqueness-db-constraint",
  "scope-leadform-getbyperiod",
  "map-domain-errors",
];

const SCHEMA = {
  type: "object",
  required: ["changeId", "validated", "applied", "rejected"],
  properties: {
    changeId: { type: "string" },
    validated: { type: "boolean" },
    applied: {
      type: "array",
      items: {
        type: "object",
        required: ["issue", "how"],
        properties: { issue: { type: "string" }, how: { type: "string" } },
      },
    },
    rejected: {
      type: "array",
      items: {
        type: "object",
        required: ["issue", "why"],
        properties: { issue: { type: "string" }, why: { type: "string" } },
      },
    },
    notes: { type: "string" },
  },
};

const GUARD = `
Treat each Codex finding as a CANDIDATE, not gospel — Codex sometimes misreads
(flags something the design already handles, or assumes code it did not read).
For each finding: open the actual artifact and the cited code, CONFIRM it is real
and not already addressed, THEN apply. Reject a finding that is wrong or already
handled and record why in "rejected". Several findings flag CONFLICTS WITH EXISTING
DURABLE SPECS (e.g. dependency-currency requiring Bull Board requireHead
byte-identical) — when real, resolve by adding a MODIFIED delta against that
capability in this change, not by ignoring it. Respect repo conventions: hardcoded
Russian is only a bug in apps/web JSX; ComingSoon is intentional; a single init
migration is intentional (MVP — reshape schema.prisma + hand-edit the init
migration, NOT an incremental alter-table migration; do not tell implementers to
run pnpm lint:md, it is intentionally removed); English-only prod UI is intentional.
You edit the OpenSpec change ARTIFACTS only — do NOT modify product code and do NOT
mark tasks done.`;

phase("Revise");
const results = await parallel(
  CHANGES.map((id) => () =>
    agent(
      `Revise ONE OpenSpec change in the CustomReport repo by applying the valid
adversarial-review findings. You edit only openspec/changes/${id}/ artifacts.

CHANGE: ${id}

STEP 1 — get the findings:
Read ${REVIEW_FILE} (a JSON array). Find the object whose changeId === "${id}". Its
"issues" array holds the Codex findings (severity/where/problem/fix).

STEP 2 — verify + apply (candidates, not gospel):
For each finding, open the cited artifact in openspec/changes/${id}/ and the cited
product code. If the finding is REAL and not already handled, apply the fix by
editing the right artifact: a missing requirement/scenario -> specs/**/spec.md; a
design gap -> design.md; a missing/▢ task -> tasks.md; scope/why -> proposal.md. If
the finding is a conflict with an existing durable spec and is real, add the needed
MODIFIED delta against that capability inside this change. Keep delta-spec rules:
"## ADDED/MODIFIED Requirements", "### Requirement:" with the SHALL clause on the
first body line, every requirement has a "#### Scenario:" with GIVEN/WHEN/THEN. If a
finding is wrong or already covered, do NOT edit — record it in "rejected" with the
reason.

STEP 3 — validate:
Run \`openspec validate "${id}" --strict\` and fix until it exits 0.

Return {changeId, validated, applied:[{issue,how}], rejected:[{issue,why}], notes}.
${GUARD}`,
      {
        label: `revise:${id}`,
        phase: "Revise",
        model: "opus",
        effort: "high",
        schema: SCHEMA,
      },
    ),
  ),
);

const done = results.filter(Boolean);
log(
  `Revised ${done.length}/${CHANGES.length}; validated ${done.filter((r) => r.validated).length}; ` +
    `applied ${done.reduce((n, r) => n + (r.applied?.length || 0), 0)}, ` +
    `rejected ${done.reduce((n, r) => n + (r.rejected?.length || 0), 0)}.`,
);
return { revised: done };
