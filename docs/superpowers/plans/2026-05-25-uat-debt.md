### Task 2.3: Fix form initialization

- [ ] Open the dialog component file
- [ ] Locate `useForm` / `reset` calls
- [ ] Guard the reset: only reset when dialog opens AND item data is stable (not on every render):

  ```typescript
  const form = useForm<ItemFormValues>({
    defaultValues: itemToFormValues(item),
  });

      form.reset(itemToFormValues(item));
    }
  }, [open, item?.id]); // NOT item object itself (reference instability)
  ```

  ```typescript
    await page.goto("http://localhost:3000");
    const cls = await page.evaluate(
      () =>
        new Promise<number>((resolve) => {
          let total = 0;
          new PerformanceObserver((list) => {
            list.getEntries().forEach((e: any) => {
              total += e.value;
            });
          }).observe({ type: "layout-shift", buffered: true });
          setTimeout(() => resolve(total), 2000);
        }),
    );
  ```
