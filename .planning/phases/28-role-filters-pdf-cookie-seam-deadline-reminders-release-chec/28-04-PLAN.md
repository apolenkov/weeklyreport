    Step 3.2 — Create `tools/handoff-sync.ts`:

    ```typescript
    /**
     * REL-05 — handoff-sync tool
     *
     * Reads `/CHECKLIST.md`, extracts unticked items (`- [ ]` lines), and
     * rewrites the `## Pending Manual Smoke` section in `/HANDOFF.md`.
     *
     * Idempotent: rerunning produces the same output. Section is replaced in
     * place, bounded by `&lt;!-- handoff-sync:start --&gt;` and `&lt;!-- handoff-sync:end --&gt;`
     * markers; if absent, the section is appended at end of HANDOFF.md.
     *
     * Usage: `pnpm tsx tools/handoff-sync.ts`
     */
    import { readFileSync, writeFileSync, existsSync } from 'node:fs';
    import { resolve } from 'node:path';

    const CHECKLIST_PATH = resolve(process.cwd(), 'CHECKLIST.md');
    const HANDOFF_PATH = resolve(process.cwd(), 'HANDOFF.md');
    const START_MARKER = '&lt;!-- handoff-sync:start --&gt;';
    const END_MARKER = '&lt;!-- handoff-sync:end --&gt;';

    export interface ChecklistItem {
      id: string;
      gate: 'manual' | 'CI';
      description: string;
    }

    export function parseUntickedItems(checklist: string): ChecklistItem[] {
      const items: ChecklistItem[] = [];
      const re = /^- \[ \] ([A-Z]+-[A-Z]+-\d{3}) \[(manual|CI)\] — (.+)$/gm;
      let m: RegExpExecArray | null;
      while ((m = re.exec(checklist)) !== null) {
        items.push({ id: m[1]!, gate: m[2] as 'manual' | 'CI', description: m[3]! });
      }
      return items;
    }

    export function buildSection(items: ChecklistItem[]): string {
      const manual = items.filter((i) =&gt; i.gate === 'manual');
      const ci = items.filter((i) =&gt; i.gate === 'CI');
      const lines: string[] = [];
      lines.push('## Pending Manual Smoke');
      lines.push('');
      lines.push('Auto-generated by `pnpm gsd:checklist:sync` from `/CHECKLIST.md`.');
      lines.push('');
      lines.push(`### Manual (${manual.length})`);
      for (const item of manual) lines.push(`- [ ] **${item.id}** — ${item.description}`);
      lines.push('');
      lines.push(`### CI-gated (${ci.length})`);
      for (const item of ci) lines.push(`- [ ] **${item.id}** — ${item.description}`);
      lines.push('');
      return lines.join('\n');
    }

    export function rewriteHandoff(handoff: string, generatedSection: string): string {
      const block = `${START_MARKER}\n${generatedSection}${END_MARKER}\n`;
      const startIdx = handoff.indexOf(START_MARKER);
      const endIdx = handoff.indexOf(END_MARKER);
      if (startIdx !== -1 &amp;&amp; endIdx !== -1 &amp;&amp; endIdx &gt; startIdx) {
        return handoff.slice(0, startIdx) + block + handoff.slice(endIdx + END_MARKER.length + 1);
      }
      // append
      const sep = handoff.endsWith('\n') ? '\n' : '\n\n';
      return handoff + sep + block;
    }

    export function main(): void {
      if (!existsSync(CHECKLIST_PATH)) {
        process.stderr.write(`handoff-sync: CHECKLIST.md not found at ${CHECKLIST_PATH}\n`);
        process.exit(1);
      }
      if (!existsSync(HANDOFF_PATH)) {
        process.stderr.write(`handoff-sync: HANDOFF.md not found at ${HANDOFF_PATH}\n`);
        process.exit(1);
      }
      const checklist = readFileSync(CHECKLIST_PATH, 'utf8');
      const handoff = readFileSync(HANDOFF_PATH, 'utf8');
      const items = parseUntickedItems(checklist);
      const section = buildSection(items);
      const next = rewriteHandoff(handoff, section);
      if (next !== handoff) {
        writeFileSync(HANDOFF_PATH, next);
        process.stdout.write(`handoff-sync: rewrote ${HANDOFF_PATH} (${items.length} items)\n`);
      } else {
        process.stdout.write(`handoff-sync: HANDOFF.md already up to date (${items.length} items)\n`);
      }
    }

    if (process.argv[1] &amp;&amp; process.argv[1].endsWith('handoff-sync.ts')) {
      main();
    }
    ```

    Step 3.3 — Create `tools/handoff-sync.test.ts`:

    ```typescript
    import { describe, it, expect } from 'vitest';
    import { parseUntickedItems, buildSection, rewriteHandoff } from './handoff-sync';

    describe('handoff-sync', () =&gt; {
      it('parses CI and manual items', () =&gt; {
        const md = [
          '## Section',
          '- [ ] AUTH-LOGIN-001 [manual] — Login as lead',
          '- [ ] AUTH-FILTERS-001 [CI] — Role filter tests',
          '- [x] DONE-001 [manual] — already done', // ignored
        ].join('\n');
        const items = parseUntickedItems(md);
        expect(items).toEqual([
          { id: 'AUTH-LOGIN-001', gate: 'manual', description: 'Login as lead' },
          { id: 'AUTH-FILTERS-001', gate: 'CI', description: 'Role filter tests' },
        ]);
      });

      it('buildSection groups manual and CI', () =&gt; {
        const out = buildSection([
          { id: 'A-B-001', gate: 'manual', description: 'm' },
          { id: 'A-B-002', gate: 'CI', description: 'c' },
        ]);
        expect(out).toContain('### Manual (1)');
        expect(out).toContain('### CI-gated (1)');
      });

      it('rewrites between markers idempotently', () =&gt; {
        const original = [
          '# Handoff',
          '',
          '&lt;!-- handoff-sync:start --&gt;',
          'OLD CONTENT',
          '&lt;!-- handoff-sync:end --&gt;',
          'tail',
        ].join('\n');
        const next = rewriteHandoff(original, 'NEW SECTION\n');
        expect(next).toContain('NEW SECTION');
        expect(next).not.toContain('OLD CONTENT');
        expect(next).toContain('tail');
        // running again is a no-op
        const next2 = rewriteHandoff(next, 'NEW SECTION\n');
        expect(next2).toBe(next);
      });

      it('appends when markers absent', () =&gt; {
        const original = '# Handoff\n\nbody';
        const next = rewriteHandoff(original, 'GEN\n');
        expect(next).toMatch(/&lt;!-- handoff-sync:start --&gt;/);
        expect(next).toMatch(/GEN/);
      });
    });
    ```
