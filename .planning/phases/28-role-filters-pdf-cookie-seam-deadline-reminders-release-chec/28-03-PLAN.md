    ```typescript
    import { Injectable } from '@nestjs/common';
    import { Cron, CronExpression } from '@nestjs/schedule';

    import type { RemindersService } from './reminders.service';
    import type { RemindersRepository } from './reminders.service';

    @Injectable()
    export class RemindersScheduler {
      constructor(
        private readonly service: RemindersService,
        private readonly repository: RemindersRepository,
      ) {}

      // NOTIF-04 — hourly tick. CronExpression.EVERY_HOUR === '0 * * * *'.
      @Cron(CronExpression.EVERY_HOUR)
      async dispatchDueDeadlineReminders(): Promise&lt;void&gt; {
        const now = new Date();
        const dueRules = await this.repository.findDueDeadlineRules(now);

        for (const rule of dueRules) {
          for (const recipient of rule.recipients) {
            const cycleDate = formatUtcDate(rule.fireAt); // YYYY-MM-DD
            await this.service.requestReminderExecution(
              rule.id,
              recipient.email,
              cycleDate,
              {
                kind: 'deadline',
                leadName: recipient.name,
                reportPeriod: rule.reportPeriodLabel,
                deadlineIso: rule.fireAt.toISOString(),
                sectionUrl: buildSectionUrl(recipient.userId, rule.reportId),
              },
            );
          }
        }
      }
    }

    function formatUtcDate(d: Date): string {
      return d.toISOString().slice(0, 10);
    }

    function buildSectionUrl(_userId: string, reportId: string): string {
      const base = process.env['APP_BASE_URL'] ?? 'http://localhost:3000';
      return `${base}/lead/report?reportId=${reportId}`;
    }
    ```
