**1. Rewrite `apps/web/src/app/(app)/lead/report/page.tsx`:**

Remove ALL imports from: `section-storage`, `demo-section`, `dev-session`, `use-browser-state`.

New implementation pattern:
```typescript
'use client';

import { trpc } from '../../../../lib/trpc';
import { QueryError } from '../../../../components/common/query-error';
import { AppShell } from '../../../../components/navigation/app-shell';
import { ReportSectionForm } from '../../../../components/report-section/report-section-form';
import type { DraftSavePayload, ReportSectionReadModel, SubmitSectionPayload } from '@customreport/shared';

export default function LeadReportPage() {
  const sectionQuery = trpc.reports.getActive.useQuery();
  const saveDraftMutation = trpc.reports.saveDraft.useMutation();
  const submitMutation = trpc.reports.submit.useMutation();

  if (sectionQuery.isLoading) return <p>Loading...</p>;
  if (sectionQuery.isError) return <QueryError error={sectionQuery.error} onRetry={() => sectionQuery.refetch()} />;

  const section = sectionQuery.data;

  async function handleSaveDraft(input: DraftSavePayload): Promise<ReportSectionReadModel> {
    const result = await saveDraftMutation.mutateAsync(input);
    return result;
  }

  async function handleSubmit(input: SubmitSectionPayload): Promise<ReportSectionReadModel> {
    const result = await submitMutation.mutateAsync(input);
    return result;
  }

  return (
    <AppShell activeReportId={null} role="lead" currentPath="/lead/report">
      <ReportSectionForm
        initialSection={section}
        onSaveDraft={handleSaveDraft}
        onSubmit={handleSubmit}
      />
    </AppShell>
  );
}
```

Note: `AppShell` props `activeReportId` and `role` — since we no longer have the dev session, temporarily pass `null` and `'lead'`. These will come from a real session context or the tRPC data. The `section` data from `getActive` includes the reportId so `activeReportId` can be derived: `section.reportId`. Check AppShell props to see what it actually needs and pass accordingly.

```typescript
'use client';

import { trpc } from '../../../../lib/trpc';
import { QueryError } from '../../../../components/common/query-error';
import { AppShell } from '../../../../components/navigation/app-shell';
import { AssembledReportReview } from '../../../../components/head/assembled-report-review';
import type { DraftSavePayload } from '@customreport/shared';

export default function HeadReportPage() {
  const reportQuery = trpc.review.assembledReport.useQuery();
  const publishMutation = trpc.review.publish.useMutation();
  const utils = trpc.useUtils();

  if (reportQuery.isLoading) return <p>Loading...</p>;
  if (reportQuery.isError) return <QueryError error={reportQuery.error} onRetry={() => reportQuery.refetch()} />;

  async function handleUpdateSection(input: DraftSavePayload) {
    // Head editing a section in review — use saveDraft mutation
    // This delegates to the same saveDraft endpoint the lead uses
    // The assembled report is then re-fetched
    return reportQuery.data;
  }

  async function handlePublish() {
    const result = await publishMutation.mutateAsync();
    await utils.review.assembledReport.invalidate();
    return result;
  }

  return (
    <AppShell activeReportId={null} role="head" currentPath="/head/report">
      <AssembledReportReview
        initialReport={reportQuery.data}
        onPublish={handlePublish}
        onUpdateSection={handleUpdateSection}
      />
    </AppShell>
  );
}
```

The backend `review.assembledReport` query returns the full assembled report. For section edits during review, the head would call `reports.saveDraft` for the specific section. Read the AssembledReportReview component to understand what `onUpdateSection` actually does with its return value and wire it correctly. **1. Rewrite `apps/web/src/app/(app)/reports/[reportId]/page.tsx`:**

Remove ALL imports from: `report-runtime`, `dev-session`, `use-browser-state`.

```typescript
'use client';

import { useParams } from 'next/navigation';
import { useState } from 'react';
import { trpc } from '../../../../lib/trpc';
import { QueryError } from '../../../../components/common/query-error';
import { AppShell } from '../../../../components/navigation/app-shell';
import { PublishedReport } from '../../../../components/report/published-report';

export default function PublishedReportPage() {
  const params = useParams<{ reportId: string }>();
  const reportId = params?.reportId ?? '';

  const reportQuery = trpc.archive.detail.useQuery(
    { reportId },
    { enabled: Boolean(reportId) },
  );
  const pdfRequestMutation = trpc.pdfExport.request.useMutation();
  const [pdfJobId, setPdfJobId] = useState<string | null>(null);
  const pdfStatusQuery = trpc.pdfExport.status.useQuery(
    { jobId: pdfJobId! },
    { enabled: Boolean(pdfJobId), refetchInterval: 2000 },
  );

  if (reportQuery.isLoading) return <p>Loading...</p>;
  if (reportQuery.isError) return <QueryError error={reportQuery.error} onRetry={() => reportQuery.refetch()} />;

  function handleRequestPdfExport() {
    if (!reportId) return;
    pdfRequestMutation.mutate(
      { reportId },
      {
        onSuccess: (job) => {
          setPdfJobId(job.id);
        },
      },
    );
  }

  // Read the published-report.tsx component to understand the exact shape of
  // the `report` and `exportJob` props. The archive.detail query returns an
  // object that should contain the `report` field. Pass accordingly:
  // - reportQuery.data.report (or reportQuery.data if the type matches directly)
  // - pdfStatusQuery.data for exportJob (or null if no PDF requested)
  
  const reportData = reportQuery.data;
  const exportJob = pdfStatusQuery.data ?? pdfRequestMutation.data ?? null;

  return (
    <AppShell
      activeReportId={reportId}
      currentPath={`/reports/${reportId}`}
      role="viewer"
    >
      {reportData ? (
        <PublishedReport
          exportJob={exportJob}
          onRequestPdfExport={handleRequestPdfExport}
          report={reportData}
        />
      ) : (
        <p>Published report not found.</p>
      )}
    </AppShell>
  );
}
```

The PDF polling uses `refetchInterval: 2000` (every 2 seconds) when a jobId is set. This stops automatically when the component unmounts or when the job status is `completed` (the executor should check the PdfExportJob type and add a condition to stop polling when status is 'completed' or 'failed').

Read the `PublishedReport` component and the `ArchiveService.getArchiveReport()` return type to ensure the data shape matches the `report` prop type. The `report` prop expects `AssembledReport` — if `archive.detail` returns a wrapper object, extract the correct field.
