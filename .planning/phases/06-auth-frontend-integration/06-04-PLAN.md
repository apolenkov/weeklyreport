The admin page is the most complex page with 8 callback handlers. Each handler maps to a tRPC mutation:

```typescript
'use client';

import { trpc } from '../../../lib/trpc';
import { QueryError } from '../../../components/common/query-error';
import { AppShell } from '../../../components/navigation/app-shell';
import { AdminPanel } from '../../../components/admin/admin-panel';
import type { AdminUser, AdminProject, ReportPeriod, AdminReminderRule } from '@customreport/shared';

export default function AdminPage() {
  const snapshotQuery = trpc.admin.snapshot.useQuery();
  const utils = trpc.useUtils();

  const saveUserMutation = trpc.admin.saveUser.useMutation({ onSuccess: () => utils.admin.snapshot.invalidate() });
  const deleteUserMutation = trpc.admin.deleteUser.useMutation({ onSuccess: () => utils.admin.snapshot.invalidate() });
  const saveProjectMutation = trpc.admin.saveProject.useMutation({ onSuccess: () => utils.admin.snapshot.invalidate() });
  const savePeriodMutation = trpc.admin.savePeriod.useMutation({ onSuccess: () => utils.admin.snapshot.invalidate() });
  const saveReminderRuleMutation = trpc.admin.saveReminderRule.useMutation({ onSuccess: () => utils.admin.snapshot.invalidate() });
  const executeReminderMutation = trpc.reminders.execute.useMutation({ onSuccess: () => utils.admin.snapshot.invalidate() });
  const carryOverMutation = trpc.carryOver.execute.useMutation();

  if (snapshotQuery.isLoading) return <p>Loading...</p>;
  if (snapshotQuery.isError) return <QueryError error={snapshotQuery.error} onRetry={() => snapshotQuery.refetch()} />;

  const snapshot = snapshotQuery.data;

  async function handleUpsertUser(user: AdminUser) {
    await saveUserMutation.mutateAsync(user);
    // Refetch snapshot to get latest state
    const fresh = await utils.admin.snapshot.fetch();
    return fresh;
  }

  async function handleRemoveUser(userId: string) {
    await deleteUserMutation.mutateAsync({ userId });
    const fresh = await utils.admin.snapshot.fetch();
    return fresh;
  }

  async function handleUpsertProject(project: AdminProject) {
    await saveProjectMutation.mutateAsync(project);
    const fresh = await utils.admin.snapshot.fetch();
    return fresh;
  }

  async function handleUpsertPeriod(period: ReportPeriod) {
    await savePeriodMutation.mutateAsync(period);
    const fresh = await utils.admin.snapshot.fetch();
    return fresh;
  }

  async function handleUpsertReminderRule(rule: AdminReminderRule) {
    await saveReminderRuleMutation.mutateAsync(rule);
    const fresh = await utils.admin.snapshot.fetch();
    return fresh;
  }

  async function handleRequestReminderExecution(ruleId: string, recipientEmail: string) {
    await executeReminderMutation.mutateAsync({ ruleId, recipientEmail });
    const fresh = await utils.admin.snapshot.fetch();
    return fresh;
  }

  async function handleRunCarryOver() {
    // CarryOver needs sourceReportId and targetReportId
    // Read AdminPanel component to understand how it calls this
    // The original code used the last period from snapshot.periods
    // For the tRPC version, we need both IDs. Extract from snapshot:
    const periods = snapshot.periods;
    if (periods.length < 2) return false;
    // Source is the second-to-last, target is the last period
    const sourceReportId = periods[periods.length - 2]?.id;
    const targetReportId = periods[periods.length - 1]?.id;
    if (!sourceReportId || !targetReportId) return false;

    try {
      await carryOverMutation.mutateAsync({ sourceReportId, targetReportId });
      return true;
    } catch {
      return false;
    }
  }

  return (
    <AppShell activeReportId={null} currentPath="/admin" role="head">
      <AdminPanel
        initialSnapshot={snapshot}
        onRemoveUser={handleRemoveUser}
        onRunCarryOver={handleRunCarryOver}
        onRequestReminderExecution={handleRequestReminderExecution}
        onUpsertPeriod={handleUpsertPeriod}
        onUpsertProject={handleUpsertProject}
        onUpsertReminderRule={handleUpsertReminderRule}
        onUpsertUser={handleUpsertUser}
      />
    </AppShell>
  );
}
```

IMPORTANT: Read the `AdminPanel` component to confirm the exact prop types. Each `handle*` function must return what `AdminPanel` expects (likely `AdminSnapshot`). The `utils.admin.snapshot.fetch()` call refetches the snapshot from the API and returns the data, ensuring the component gets fresh state after each mutation.

Also check if `AdminPanel` uses `onRunCarryOver` with specific arguments — the original used `snapshot.periods` to find the target. Adjust carry-over logic accordingly.
