    **Create a11y/accessibility.spec.ts:**
    ```typescript
    import { test, expect } from '@playwright/test';
    import AxeBuilder from '@axe-core/playwright';

    const pagesToAudit = [
      { name: 'lead-dashboard', path: '/lead', role: 'lead' },
      { name: 'lead-section-form', path: '/lead/section', role: 'lead' },
      { name: 'head-dashboard', path: '/head', role: 'head' },
      { name: 'head-review', path: '/head/review', role: 'head' },
      { name: 'admin-panel', path: '/admin', role: 'head' },
      { name: 'archive', path: '/archive', role: 'viewer' },
    ];

    for (const page of pagesToAudit) {
      test(`a11y audit: ${page.name}`, async ({ page: p }) => {
        await p.goto(`/auth/dev-login?role=${page.role}`);
        await p.waitForURL('**/*');
        await p.goto(page.path);
        await p.waitForLoadState('networkidle');

        const results = await new AxeBuilder({ page: p })
          .withTags(['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa'])
          .analyze();

        expect(results.violations).toEqual([]);
      });
    }

    test('a11y: keyboard navigation through lead form', async ({ page }) => {
      await page.goto('/auth/dev-login?role=lead');
      await page.waitForURL('**/*');
      await page.goto('/lead/section');
      await page.waitForLoadState('networkidle');

      // Tab through form elements and verify focus order
      await page.keyboard.press('Tab');
      const firstFocused = await page.evaluate(() => document.activeElement?.tagName);
      expect(firstFocused).toBeTruthy();

      // Verify all form inputs are reachable via Tab
      const focusableCount = await page.evaluate(() => {
        const els = document.querySelectorAll('input, select, textarea, button, [tabindex="0"]');
        return els.length;
      });
      expect(focusableCount).toBeGreaterThan(5);
    });
    ```
    The audit uses WCAG 2.1 AA rules. Any violation fails the test with details about which element failed and which rule was broken.

    ```typescript
    import { test, expect } from '@playwright/test';

    const breakpoints = [
      { name: 'mobile-small', width: 320, height: 568 },
      { name: 'mobile', width: 375, height: 812 },
      { name: 'tablet', width: 768, height: 1024 },
      { name: 'desktop', width: 1024, height: 768 },
      { name: 'desktop-wide', width: 1920, height: 1080 },
    ];

    const keyPages = [
      { name: 'lead-form', path: '/lead/section', role: 'lead' },
      { name: 'head-dashboard', path: '/head', role: 'head' },
      { name: 'admin-panel', path: '/admin', role: 'head' },
    ];

    for (const bp of breakpoints) {
      for (const page of keyPages) {
        test(`responsive ${bp.name}: ${page.name}`, async ({ page: p }) => {
          await p.setViewportSize({ width: bp.width, height: bp.height });
          await p.goto(`/auth/dev-login?role=${page.role}`);
          await p.waitForURL('**/*');
          await p.goto(page.path);
          await p.waitForLoadState('networkidle');

          // Verify no horizontal overflow
          const bodyWidth = await p.evaluate(() => document.body.scrollWidth);
          expect(bodyWidth).toBeLessThanOrEqual(bp.width + 1); // +1 for rounding

          // Verify main content is visible
          const mainContent = p.locator('main');
          if (await mainContent.count() > 0) {
            await expect(mainContent).toBeVisible();
          }

          // On mobile, verify mobile nav is present instead of desktop nav
          if (bp.width < 768) {
            const mobileNavTrigger = p.locator('[data-testid="mobile-nav-trigger"], button[aria-label*="menu"]');
            // Mobile nav should be available (either visible trigger or hamburger)
          }

          // Screenshot for visual comparison
          await expect(p).toHaveScreenshot(`${page.name}-${bp.name}.png`, {
            fullPage: true,
            animations: 'disabled',
          });
        });
      }
    }

    test('responsive: mobile nav opens and closes', async ({ page }) => {
      await page.setViewportSize({ width: 375, height: 812 });
      await page.goto('/auth/dev-login?role=lead');
      await page.waitForURL('**/*');
      await page.goto('/lead');
      await page.waitForLoadState('networkidle');

      // Find and click mobile nav trigger
      const trigger = page.locator('button').filter({ hasText: /menu/i }).or(
        page.locator('[data-testid="mobile-nav-trigger"]')
      );
      if (await trigger.count() > 0) {
        await trigger.click();
        // Verify nav sheet is visible
        const navSheet = page.locator('[role="dialog"]').or(page.locator('nav'));
        await expect(navSheet).toBeVisible();
      }
    });
    ```

    This creates 15 responsive tests (5 breakpoints x 3 key pages) plus 1 mobile nav interaction test.
    Each test verifies:
    1. No horizontal overflow (content fits viewport)
    2. Main content area is visible
    3. Visual screenshot for manual comparison
