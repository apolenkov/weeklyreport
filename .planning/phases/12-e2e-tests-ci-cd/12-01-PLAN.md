**1b. Fix tRPC context to read JWT cookie even in dev mode** in `apps/api/src/infrastructure/trpc/trpc.context.ts`:

The current code returns a hardcoded DEV_SESSION when AUTH_DEV_MODE=true, ignoring any JWT cookie. This makes per-role E2E impossible. Change the logic so that in dev mode, if a valid JWT cookie exists, it is used; otherwise fall back to the DEV_SESSION.

Replace the context factory body:

```typescript
export function createContextFactory(_app: INestApplication) {
  const authController = createAuthController(process.env);

  return async ({ req }: CreateExpressContextOptions): Promise<TrpcContext> => {
    const token = req.cookies?.['customreport_session'] as string | undefined;

    if (token) {
      try {
        const session = authController.verifySessionToken(token);
        return { session };
      } catch {
        // Token invalid/expired -- fall through
      }
    }

    // In dev mode without a valid cookie, use fallback session
    if (process.env['AUTH_DEV_MODE'] === 'true') {
      return { session: DEV_SESSION };
    }

    return { session: null };
  };
}
```

Keep the DEV_SESSION constant (still useful as fallback for manual dev browsing).

**2c. Create `apps/web/e2e/auth.setup.ts`** -- global auth setup that hits dev-login for each role and saves storageState:

```typescript
import { test as setup, expect } from '@playwright/test';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const authDir = path.join(__dirname, '.auth');

const API_URL = 'http://127.0.0.1:3101';

const roles = ['lead', 'head', 'viewer'] as const;

for (const role of roles) {
  setup(`authenticate as ${role}`, async ({ page }) => {
    // Hit the dev-login endpoint -- it creates user in DB and sets JWT cookie
    await page.goto(`${API_URL}/auth/dev-login?role=${role}`);

    // After redirect, verify cookie was set by checking we can access a page
    await expect(page).toHaveURL(/localhost:3100/);

    // Save storageState (cookies + localStorage) for this role
    await page.context().storageState({
      path: path.join(authDir, `${role}.json`),
    });
  });
}
```

import { test as base, expect, type Page, type BrowserContext } from '@playwright/test';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const authDir = path.join(__dirname, '.auth');

type RoleFixtures = {
  leadPage: Page;
  headPage: Page;
  viewerPage: Page;
};

export const test = base.extend<RoleFixtures>({
  leadPage: async ({ browser }, use) => {
    const context = await browser.newContext({
      storageState: path.join(authDir, 'lead.json'),
    });
    const page = await context.newPage();
    await use(page);
    await context.close();
  },

  headPage: async ({ browser }, use) => {
    const context = await browser.newContext({
      storageState: path.join(authDir, 'head.json'),
    });
    const page = await context.newPage();
    await use(page);
    await context.close();
  },

  viewerPage: async ({ browser }, use) => {
    const context = await browser.newContext({
      storageState: path.join(authDir, 'viewer.json'),
    });
    const page = await context.newPage();
    await use(page);
    await context.close();
  },
});

export { expect };
