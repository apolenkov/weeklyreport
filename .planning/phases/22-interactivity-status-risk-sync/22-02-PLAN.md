type SyncSource = 'user' | 'auto-sync';

function useStatusRiskSync({ products, risks, onProductsChange, onRisksChange }) {
  const syncSourceRef = useRef<SyncSource>('user');

  const handleStageStatusChange = useCallback((productId, stageName, newStatus, source: SyncSource = 'user') => {
    onProductsChange(prev => updateStage(prev, productId, stageName, newStatus));
    if (source === 'user' && (newStatus === 'at-risk' || newStatus === 'blocked')) {
      const exists = risks.some(r => r.productId === productId && r.stageName === stageName);
      if (!exists) {
        syncSourceRef.current = 'auto-sync';
        onRisksChange(prev => [...prev, { id: nanoid(), productId, productName: lookupName(products, productId), stageName, severity: newStatus === 'blocked' ? 'blocked' : 'at-risk', text: '', impact: '', mitigation: '' }]);
        syncSourceRef.current = 'user';
      }
    }
  }, [risks, products, onProductsChange, onRisksChange]);

  const handleRiskFilled = useCallback((riskId, source: SyncSource = 'user') => {
    // Only fire suggestion when source==='user' AND the bound stage is 'on-track'
    // ... implementation invokes a callback that the bridge wires to a confirm dialog
  }, [...]);

  return { handleStageStatusChange, handleRiskFilled, syncSource: syncSourceRef.current };
}

       'use client';
       import { useState } from 'react';
       import { useStatusRiskSync } from '../../hooks/use-status-risk-sync';
       import { SdlcPipeline } from './sdlc-pipeline';
       import { RisksSection } from './risks-section';
       import type { Product, RiskEntry } from './mock-data';

       export type LeadFormSyncBridgeProps = {
         readonly products: Product[];
         readonly risks: RiskEntry[];
         readonly onProductsChange: (next: Product[]) => void;
         readonly onRisksChange: (next: RiskEntry[]) => void;
         readonly readOnly?: boolean;
       };

       export function LeadFormSyncBridge({ products, risks, onProductsChange, onRisksChange, readOnly }: LeadFormSyncBridgeProps) {
         const [pendingSuggestion, setPendingSuggestion] = useState<{ productId: string; productName: string; stageName: string; suggestedStatus: 'at-risk' | 'blocked' } | null>(null);

         const sync = useStatusRiskSync({
           products,
           risks,
           onProductsChange: (updater) => onProductsChange(typeof updater === 'function' ? (updater as any)(products) : updater),
           onRisksChange: (updater) => onRisksChange(typeof updater === 'function' ? (updater as any)(risks) : updater),
           requestStatusUpdateConfirmation: setPendingSuggestion,
         });

         return (
           <>
             <SdlcPipeline
               products={products}
               readOnly={readOnly}
               onStatusChange={(productId, stageName, status) => !readOnly && sync.handleStageStatusChange(productId, stageName, status, 'user')}
             />
             <RisksSection
               risks={risks}
               readOnly={readOnly}
               onChange={(next) => !readOnly && onRisksChange(next)}
               onRiskFilled={(riskId) => !readOnly && sync.handleRiskFilled(riskId, 'user')}
             />
             {pendingSuggestion && (
               <ConfirmDialog
                 title="Update stage status?"
                 message={`A risk was added for ${pendingSuggestion.productName} / ${pendingSuggestion.stageName}. Would you like to mark this stage as ${pendingSuggestion.suggestedStatus}?`}
                 onConfirm={() => { sync.applyStatusSuggestion(pendingSuggestion.productId, pendingSuggestion.stageName, pendingSuggestion.suggestedStatus); setPendingSuggestion(null); }}
                 onCancel={() => setPendingSuggestion(null)}
               />
             )}
           </>
         );
       }
