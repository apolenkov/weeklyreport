    'use client';

    import { createContext, useContext, useMemo, type ReactNode } from 'react';
    import { useQuery } from '@tanstack/react-query';
    import { useTRPC } from './trpc-context'; // existing helper exporting useTRPC()

    type SessionUser = {
      id: string;
      name: string;
      email: string;
      role: 'lead' | 'head' | 'viewer';
    };

    type SessionState = {
      user: SessionUser | null;
      isLoading: boolean;
      isAuthenticated: boolean;
    };

    const SessionContext = createContext<SessionState | null>(null);

    export function SessionProvider({ children }: { children: ReactNode }) {
      const trpc = useTRPC();
      // auth.me returns UserSession | null — we map to SessionUser shape.
      const { data, isLoading } = useQuery(trpc.auth.me.queryOptions());

      const value = useMemo<SessionState>(() => {
        if (!data) {
          return { user: null, isLoading, isAuthenticated: false };
        }
        return {
          user: {
            id: data.userId,
            name: data.name,
            email: data.email,
            role: data.role,
          },
          isLoading: false,
          isAuthenticated: true,
        };
      }, [data, isLoading]);

      return <SessionContext.Provider value={value}>{children}</SessionContext.Provider>;
    }

    export function useSession(): SessionState {
      const ctx = useContext(SessionContext);
      if (!ctx) {
        throw new Error('useSession must be used within <SessionProvider>');
      }
      return ctx;
    }

    Add a custom link BEFORE httpBatchLink. The tRPC link API is `({ next, op }) => observable(...)`. On error from downstream, inspect for httpStatus 401 / TRPC code UNAUTHORIZED, then redirect.

    ```typescript
    import { createTRPCClient as createTRPCClientCore, httpBatchLink, TRPCClientError, type TRPCLink } from '@trpc/client';
    import { observable } from '@trpc/server/observable';
    // ...existing imports

    const AUTH_REDIRECT_EXEMPT = /^\/(?:$|auth\/)/; // matches "/" exactly and "/auth/*"

    function createAuthRedirectLink<T extends AppRouter>(): TRPCLink<T> {
      return () => ({ next, op }) => {
        return observable((observer) => {
          const subscription = next(op).subscribe({
            next: (value) => observer.next(value),
            error: (err) => {
              if (typeof window !== 'undefined' && err instanceof TRPCClientError) {
                const httpStatus = (err.data as { httpStatus?: number } | undefined)?.httpStatus;
                const isUnauthorized = httpStatus === 401 || err.data?.code === 'UNAUTHORIZED';
                const currentPath = window.location.pathname;
                const onAuthPage = AUTH_REDIRECT_EXEMPT.test(currentPath);
                if (isUnauthorized && !onAuthPage) {
                  const next = encodeURIComponent(currentPath + window.location.search);
                  window.location.assign(`/?next=${next}`);
                  // Do not propagate; redirect is in flight.
                  return;
                }
              }
              observer.error(err);
            },
            complete: () => observer.complete(),
          });
          return () => subscription.unsubscribe();
        });
      };
    }

    export function makeTRPCClient() {
      return createTRPCClientCore<AppRouter>({
        links: [
          createAuthRedirectLink(),  // MUST come before httpBatchLink
          httpBatchLink({
            transformer: superjson,
            url: '/trpc',
            fetch: (url, options) => fetch(url, { ...options, credentials: 'include' }),
          }),
        ],
      });
    }
    ```
