    2. Create `Dockerfile.api` at monorepo root (4-stage build):
       ```dockerfile
       # Stage 1: Base with pnpm and turbo
       FROM node:20-alpine AS base
       RUN apk add --no-cache libc6-compat
       RUN corepack enable && corepack prepare pnpm@9.15.0 --activate
       RUN pnpm add -g turbo

       # Stage 2: Prune monorepo for api workspace
       FROM base AS pruner
       WORKDIR /app
       COPY . .
       RUN turbo prune @customreport/api --docker

       # Stage 3: Install deps and build
       FROM base AS builder
       WORKDIR /app
       COPY --from=pruner /app/out/pnpm-lock.yaml ./pnpm-lock.yaml
       COPY --from=pruner /app/out/pnpm-workspace.yaml ./pnpm-workspace.yaml
       COPY --from=pruner /app/out/json/ .
       RUN pnpm install --frozen-lockfile
       COPY --from=pruner /app/out/full/ .
       # Generate Prisma client
       RUN pnpm exec prisma generate
       RUN pnpm turbo build --filter=@customreport/api

       # Stage 4: Production runner
       # Use Playwright base image for PDF generation (Chromium pre-installed)
       FROM mcr.microsoft.com/playwright:v1.52.0-noble AS runner
       RUN groupadd --system --gid 1001 nodejs && \
           useradd --system --uid 1001 nodejs
       WORKDIR /app
       COPY --from=builder --chown=nodejs:nodejs /app .
       # Create PDF storage directory
       RUN mkdir -p /app/data/pdfs && chown nodejs:nodejs /app/data/pdfs
       USER nodejs
       WORKDIR /app/apps/api
       ENV NODE_ENV=production
       EXPOSE 3001
       CMD ["node", "dist/main.js"]
       ```

       NOTE: The runner stage uses the Microsoft Playwright image (not Alpine) because the PDF export processor needs Chromium. This avoids Pitfall #6 (Playwright system dependencies missing in Alpine). The exact Playwright version tag should match the version in apps/web/package.json (@playwright/test version). Check `grep playwright apps/web/package.json` to confirm version and use corresponding Docker image tag.

       ```dockerfile
       USER nextjs
       ENV NODE_ENV=production
       ENV HOSTNAME="0.0.0.0"
       EXPOSE 3000
       CMD ["node", "apps/web/server.js"]
       ```

       ```ini
       [databases]
       customreport = host=postgres port=5432 dbname=customreport

       [pgbouncer]

       listen_port = 6432
       ```

       Per DOCK-06: Transaction mode with max_prepared_statements=256 (PgBouncer 1.22+ required). This avoids Pitfall #3 (prepared statement conflicts with Prisma).

    2. Create `pgbouncer/userlist.txt`:
       ```
       "customreport" "md5_placeholder_change_in_production"
       ```
       NOTE: In production, this file must contain the MD5 hash of the PostgreSQL password. For development, the edoburu/pgbouncer image auto-generates this from DATABASE_URL.

    3. Create `docker-compose.yml` at monorepo root:
       ```yaml
       services:
         postgres:
           image: postgres:15-alpine
           environment:
             POSTGRES_USER: customreport
             POSTGRES_PASSWORD: ${DB_PASSWORD:-localdev}
             POSTGRES_DB: customreport
           volumes:
             - pgdata:/var/lib/postgresql/data
           healthcheck:
             test: ["CMD-SHELL", "pg_isready -U customreport"]
             interval: 5s
             timeout: 5s
             retries: 5
           ports:
             - "5432:5432"

         redis:
           image: redis:7-alpine
           command: ["redis-server", "--maxmemory-policy", "noeviction", "--appendonly", "yes"]
           volumes:
             - redisdata:/data
           healthcheck:
             test: ["CMD", "redis-cli", "ping"]
             interval: 5s
             timeout: 5s
             retries: 5
           ports:
             - "6379:6379"

         pgbouncer:
           image: edoburu/pgbouncer:1.22.0
           environment:
             DATABASE_URL: postgresql://customreport:${DB_PASSWORD:-localdev}@postgres:5432/customreport
             POOL_MODE: transaction
             MAX_CLIENT_CONN: 200
             DEFAULT_POOL_SIZE: 20
             MAX_PREPARED_STATEMENTS: 256
           depends_on:
             postgres:
               condition: service_healthy
           ports:
             - "6432:6432"

         migrate:
           build:
             context: .
             dockerfile: Dockerfile.api
           command: ["npx", "prisma", "migrate", "deploy"]
           environment:
             DATABASE_URL: postgresql://customreport:${DB_PASSWORD:-localdev}@postgres:5432/customreport
           depends_on:
             postgres:
               condition: service_healthy
           restart: "no"

         api:
           build:
             context: .
             dockerfile: Dockerfile.api
           environment:
             DATABASE_URL: postgresql://customreport:${DB_PASSWORD:-localdev}@pgbouncer:6432/customreport
             DIRECT_URL: postgresql://customreport:${DB_PASSWORD:-localdev}@postgres:5432/customreport
             REDIS_HOST: redis
             REDIS_PORT: 6379
             NODE_ENV: production
             FRONTEND_URL: http://localhost:3000
             AUTH_DEV_MODE: "true"
             AUTH_JWT_SECRET: ${AUTH_JWT_SECRET:-change-me-in-production-min-32-chars!!}
             PDF_BASE_URL: http://web:3000
             SMTP_HOST: ${SMTP_HOST:-}
             SMTP_PORT: ${SMTP_PORT:-587}
             SMTP_FROM: ${SMTP_FROM:-noreply@example.com}
           depends_on:
             migrate:
               condition: service_completed_successfully
             redis:
               condition: service_healthy
             pgbouncer:
               condition: service_started
           healthcheck:
             test: ["CMD-SHELL", "wget -q --spider http://localhost:3001/health/live || exit 1"]
             interval: 10s
             timeout: 5s
             retries: 3
             start_period: 30s
           ports:
             - "3001:3001"
           volumes:
             - pdfdata:/app/data/pdfs

         web:
           build:
             context: .
             dockerfile: Dockerfile.web
           environment:
             NEXT_PUBLIC_API_URL: http://localhost:3001
           depends_on:
             api:
               condition: service_healthy
           ports:
             - "3000:3000"

       volumes:
         pgdata:
         redisdata:
         pdfdata:
       ```

       Key design choices:
       - Redis uses `--maxmemory-policy noeviction` (Pitfall #4 from ROADMAP)
       - migrate service connects DIRECTLY to postgres (bypasses PgBouncer -- Pitfall #4 from RESEARCH)
       - API connects to PgBouncer for pooled connections, with DIRECT_URL to postgres for runtime Prisma needs
       - Health-based startup ordering: postgres healthy -> migrate completes -> API starts -> Web starts (DOCK-05)
       - PDF_BASE_URL points to web:3000 (internal Docker network) so Playwright can reach Next.js
       - Named volumes for data persistence (pgdata, redisdata, pdfdata)

       ```yaml
           environment:

         web:

           command: ["pnpm", "--filter=@customreport/web", "dev"]
           volumes:

             - ./apps/web/app:/app/apps/web/app
             - ./packages/shared/src:/app/packages/shared/src

             NODE_ENV: development

         migrate:
           profiles: ["init"]
       ```
       Usage: `docker compose -f docker-compose.yml -f docker-compose.dev.yml up`
       The dev override:
       - Uses `target: builder` stage (has all source code and devDeps)
       - Mounts source directories as bind volumes for hot reload
       - Overrides command to run dev scripts instead of built artifacts
       - Puts migrate behind a profile so it only runs on `docker compose --profile init up`
