    1. Create `apps/api/src/infrastructure/email/email.service.ts`:
       - Injectable NestJS service wrapping nodemailer
       - Constructor creates a transporter using `nodemailer.createTransport()`:
         ```typescript
         @Injectable()
         export class EmailService {
           private readonly transporter: nodemailer.Transporter;
           private readonly fromAddress: string;
           constructor() {
             this.fromAddress = process.env.SMTP_FROM ?? 'noreply@example.com';
             const host = process.env.SMTP_HOST;
             if (host) {
               this.transporter = nodemailer.createTransport({
                 host,
                 port: Number(process.env.SMTP_PORT ?? 587),
                 secure: false,
                 auth: process.env.SMTP_USER ? { user: process.env.SMTP_USER, pass: process.env.SMTP_PASS } : undefined,
               });
             } else {
               // Dev mode: log instead of sending
               this.transporter = nodemailer.createTransport({ jsonTransport: true });
             }
           }
           async sendMail(to: string, subject: string, text: string): Promise<void> {
             await this.transporter.sendMail({ from: this.fromAddress, to, subject, text });
           }
         }
         ```

    2. Create `apps/api/src/infrastructure/bullmq/pdf-export.processor.ts`:
       - Extends WorkerHost from `@nestjs/bullmq`
       - Decorated with `@Processor('pdf-export')`
       - Injects PdfExportService (to call markExportCompleted)
       - `process(job: Job<{ reportId: string; jobId: string }>)` method:
         a. Launch Playwright chromium browser (headless)
         b. Navigate to `${process.env.PDF_BASE_URL}/reports/${job.data.reportId}/print`
         c. Wait for `networkidle`
         d. Call `page.pdf({ format: 'A4', printBackground: true })` to get PDF buffer
         e. Write PDF buffer to `/app/data/pdfs/${job.data.jobId}.pdf` (ensure dir exists with mkdirSync recursive)
         f. Compute SHA-256 checksum of the buffer
         g. Call `this.pdfExportService.markExportCompleted(job.data.jobId, artifactUrl, checksum)`
         h. Close browser
       - Use try/finally to ensure browser closes even on error
       - `@OnWorkerEvent('failed')` handler logs the failure
