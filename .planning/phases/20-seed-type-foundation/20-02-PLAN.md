```typescript
type ItemWorkflowStatus = "done" | "in_progress" | "blocked";
type ItemType = "current" | "next";
type ItemStatus = "in-progress" | "completed" | "blocked" | "planned";

export function workflowToItemStatus(
  ws: ItemWorkflowStatus,
): Exclude<ItemStatus, "planned"> {
  switch (ws) {
    case "done":
      return "completed";
    case "in_progress":
      return "in-progress";
    case "blocked":
      return "blocked";
  }
}

export function itemStatusToWorkflow(
  status: Exclude<ItemStatus, "planned">,
): ItemWorkflowStatus {
  switch (status) {
    case "completed":
      return "done";
    case "in-progress":
      return "in_progress";
    case "blocked":
      return "blocked";
  }
}

// Derives display status from DB fields. 'planned' is computed: next-week item with no explicit status.
export function deriveItemStatus(item: {
  workflowStatus: ItemWorkflowStatus | null;
  type: ItemType;
}): ItemStatus {
  if (item.workflowStatus) {
    return workflowToItemStatus(item.workflowStatus);
  }
  // No explicit workflow status: next-week items are 'planned', current items default to 'in-progress'
  return item.type === "next" ? "planned" : "in-progress";
}
```

**Create apps/web/src/lib/mappers/lead-status.ts:**

```typescript
type ReportSectionStatus = "draft" | "submitted";
type LeadStatus = "submitted" | "draft" | "pending";

/**
 * Derives the display LeadStatus from API data.
 * 'pending' is a computed state: section exists as draft but has no items yet.
 * This status has no Prisma/Zod equivalent -- it is presentation-only.
 */
export function deriveLeadStatus(context: {
  sectionStatus: ReportSectionStatus;
  itemCount: number;
}): LeadStatus {
  if (context.sectionStatus === "submitted") {
    return "submitted";
  }
  // Draft section: 'pending' if no items yet, 'draft' if work has started
  return context.itemCount > 0 ? "draft" : "pending";
}
```
