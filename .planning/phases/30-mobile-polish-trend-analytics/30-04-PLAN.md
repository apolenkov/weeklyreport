Per the advisor reconciliation: rather than running the spec against all 7 protected pages (which would require dev-login fixtures and a full webServer stack), this plan asserts the initial-render contract on:

  <action>
    Step 2.1 — Create the spec:

    ```typescript
    import { test, expect } from '@playwright/test';
    import { devLogin } from './fixtures';

    type PageEntry = {
      readonly path: string;
      readonly auth: 'public' | 'lead' | 'head';
    };

    const PAGES: ReadonlyArray<PageEntry> = [
      // Public (no DB / no session)
      { path: '/auth/login', auth: 'public' },
      { path: '/auth/error', auth: 'public' },
      { path: '/403', auth: 'public' },
      // Lead-authenticated GET pages
      { path: '/lead/report', auth: 'lead' },
      { path: '/lead/history', auth: 'lead' },
      // Head-authenticated GET pages
      { path: '/head/dashboard', auth: 'head' },
      { path: '/reports/history', auth: 'head' },
    ];

    test.describe('Mobile — no horizontal scroll @MOB-01', () => {
      // Skip on the desktop project — only mobile-ios / mobile-android run this.
      test.skip(({}, testInfo) => testInfo.project.name === 'chromium-integration');

      for (const entry of PAGES) {
        test(`${entry.path} fits the viewport`, async ({ page, context }) => {
          if (entry.auth !== 'public') {
            await devLogin(context, entry.auth);
          }

          await page.goto(entry.path, { waitUntil: 'domcontentloaded' });

          // 1. CSS-level: html overflow-x must NOT be 'scroll' / 'auto'.
          await expect(page.locator('html')).toHaveCSS('overflow-x', /(hidden|clip|visible)/);

          // 2. Pixel-level: the document must not scroll horizontally.
          const fits = await page.evaluate(
            () => document.documentElement.scrollWidth <= window.innerWidth,
          );
          expect(fits).toBe(true);
        });
      }
    });
    ```

    Step 2.2 — Inspect `apps/web/e2e/integration/fixtures.ts` to confirm `devLogin(context, role)` signature; adjust the import if the helper has a different name (e.g. `loginAs`).
