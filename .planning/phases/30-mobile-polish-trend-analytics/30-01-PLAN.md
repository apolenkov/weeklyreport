    import {
      trafficLightHistoryInputSchema,
      trafficLightHistoryOutputSchema,
      type TrafficLightHistoryInput,
      type TrafficLightHistoryOutput,
    } from '@customreport/shared';

    export interface AnalyticsRepository {
      trafficLightHistory(input: TrafficLightHistoryInput): Promise<TrafficLightHistoryOutput>;
    }

    export class AnalyticsService {
      constructor(
        private readonly repository: AnalyticsRepository,
        private readonly cache: { getOrCompute<T>(key: string, fn: () => Promise<T>): Promise<T> },
      ) {}

      async trafficLightHistory(input: TrafficLightHistoryInput): Promise<TrafficLightHistoryOutput> {
        const parsed = trafficLightHistoryInputSchema.parse(input);
        const key = `analytics:traffic-light:${stableHash(parsed)}`;
        const result = await this.cache.getOrCompute(key, () => this.repository.trafficLightHistory(parsed));
        return trafficLightHistoryOutputSchema.parse(result);
      }
    }

    function stableHash(value: unknown): string {
      // Deterministic JSON stringify w/ sorted keys (small enough for sha-free hash use; cache layer hashes again).
      return JSON.stringify(value, Object.keys(value as object).sort());
    }
    ```

    import { Injectable } from '@nestjs/common';
    import type { TrafficLightHistoryInput, TrafficLightHistoryOutput } from '@customreport/shared';
    import { PrismaService } from '../../infrastructure/prisma/prisma.service.js';
    import type { AnalyticsRepository } from './analytics.service.js';

    @Injectable()
    export class AnalyticsPrismaRepository implements AnalyticsRepository {
      constructor(private readonly prisma: PrismaService) {}

      async trafficLightHistory(input: TrafficLightHistoryInput): Promise<TrafficLightHistoryOutput> {
        const reports = await this.prisma.report.findMany({
          where: {
            status: 'published', // PUBLISHED_FLOOR — defense in depth
            periodStart: { gte: new Date(input.periodFrom) },
            periodEnd: { lte: new Date(input.periodTo) },
            ...(input.leadId ? { sections: { some: { authorId: input.leadId } } } : {}),
          },
          include: {
            sections: { include: { projectStatuses: { include: { project: true } } } },
          },
          orderBy: { periodStart: 'asc' },
        });

        // Build period axis (unique ISO YYYY-MM-DD periodEnd labels)
        const periodLabels = Array.from(
          new Set(reports.map((r) => r.periodEnd.toISOString().slice(0, 10))),
        );

        // Project axis (unique projects across the window, optionally filtered by projectId)
        const projectMap = new Map<string, string>();
        for (const r of reports) {
          for (const s of r.sections) {
            for (const ps of s.projectStatuses) {
              if (input.projectId && ps.projectId !== input.projectId) continue;
              if (!projectMap.has(ps.projectId)) projectMap.set(ps.projectId, ps.project?.name ?? 'Unknown');
            }
          }
        }

        const rows = Array.from(projectMap.entries()).map(([projectId, projectName]) => {
          const statuses = periodLabels.map((label) => {
            const r = reports.find((rr) => rr.periodEnd.toISOString().slice(0, 10) === label);
            if (!r) return null;
            for (const s of r.sections) {
              const hit = s.projectStatuses.find((ps) => ps.projectId === projectId);
              if (hit) return hit.status as 'green' | 'yellow' | 'red';
            }
            return null;
          });
          return { projectId, projectName, statuses };
        });

        return { periods: periodLabels, rows };
      }
    }
    ```

    Step 3.1 — Create `apps/api/src/modules/analytics/analytics.cache.ts`:

    ```typescript
    import { Injectable, OnModuleDestroy } from '@nestjs/common';
    import Redis from 'ioredis';

    const TTL_SECONDS = 3600;
    const PREFIX = 'analytics:';

    export interface CacheLike {
      getOrCompute<T>(key: string, fn: () => Promise<T>): Promise<T>;
      invalidateAll(): Promise<void>;
    }

    @Injectable()
    export class AnalyticsCache implements CacheLike, OnModuleDestroy {
      private readonly redis: Redis;

      constructor() {
        this.redis = new Redis({
          host: process.env['REDIS_HOST'] ?? 'localhost',
          port: Number(process.env['REDIS_PORT'] ?? 6379),
          lazyConnect: true,
          maxRetriesPerRequest: 1,
        });
      }

      async getOrCompute<T>(key: string, fn: () => Promise<T>): Promise<T> {
        try {
          const cached = await this.redis.get(key);
          if (cached) return JSON.parse(cached) as T;
        } catch {
          // Cache failure must not break the read path; fall through to compute.
        }
        const fresh = await fn();
        try {
          await this.redis.setex(key, TTL_SECONDS, JSON.stringify(fresh));
        } catch {
          // Cache write failure is non-fatal.
        }
        return fresh;
      }

      async invalidateAll(): Promise<void> {
        try {
          let cursor = '0';
          do {
            const [next, batch] = await this.redis.scan(cursor, 'MATCH', `${PREFIX}*`, 'COUNT', 100);
            if (batch.length > 0) await this.redis.del(...batch);
            cursor = next;
          } while (cursor !== '0');
        } catch {
          // Best-effort invalidation; subsequent reads still hit TTL within the hour.
        }
      }

      async onModuleDestroy(): Promise<void> {
        await this.redis.quit().catch(() => undefined);
      }
    }
    ```

    import { describe, it, expect, vi } from 'vitest';

    function inMemoryCache(): { store: Map<string, string>; cache: import('./analytics.cache.js').CacheLike } {
      const store = new Map<string, string>();
      return {
        store,
        cache: {
          async getOrCompute<T>(key: string, fn: () => Promise<T>): Promise<T> {
            const hit = store.get(key);
            if (hit) return JSON.parse(hit) as T;
            const v = await fn();
            store.set(key, JSON.stringify(v));
            return v;
          },
          async invalidateAll() {
            for (const k of store.keys()) if (k.startsWith('analytics:')) store.delete(k);
          },
        },
      };
    }

    describe('AnalyticsCache contract', () => {
      it('hits cache on second call', async () => {
        const { cache } = inMemoryCache();
        const fn = vi.fn(async () => ({ ok: true }));
        await cache.getOrCompute('analytics:k', fn);
        await cache.getOrCompute('analytics:k', fn);
        expect(fn).toHaveBeenCalledTimes(1);
      });

      it('invalidateAll clears prefixed keys', async () => {
        const { cache, store } = inMemoryCache();
        await cache.getOrCompute('analytics:a', async () => ({ a: 1 }));
        await cache.getOrCompute('analytics:b', async () => ({ b: 2 }));
        store.set('other:keep', '1');
        await cache.invalidateAll();
        expect(store.has('analytics:a')).toBe(false);
        expect(store.has('analytics:b')).toBe(false);
        expect(store.has('other:keep')).toBe(true);
      });
    });
