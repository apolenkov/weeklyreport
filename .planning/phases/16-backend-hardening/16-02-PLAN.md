**1. Create BullMQ health indicator (apps/api/src/infrastructure/health/bullmq.health.ts):**

Create a new file following the pattern of RedisHealthIndicator:
```typescript
import { Injectable } from '@nestjs/common';
import { HealthIndicator, HealthIndicatorResult, HealthCheckError } from '@nestjs/terminus';
import { Queue } from 'bullmq';
import { InjectQueue } from '@nestjs/bullmq';

@Injectable()
export class BullMQHealthIndicator extends HealthIndicator {
  constructor(
    @InjectQueue('pdf-export') private readonly pdfQueue: Queue,
    @InjectQueue('email-reminder') private readonly emailQueue: Queue,
  ) {
    super();
  }

  async isHealthy(key: string): Promise<HealthIndicatorResult> {
    try {
      const [pdfCounts, emailCounts] = await Promise.all([
        this.pdfQueue.getJobCounts(),
        this.emailQueue.getJobCounts(),
      ]);

      return this.getStatus(key, true, {
        pdf_export: { waiting: pdfCounts.waiting, active: pdfCounts.active },
        email_reminder: { waiting: emailCounts.waiting, active: emailCounts.active },
      });
    } catch (error) {
      throw new HealthCheckError(
        `${key} health check failed`,
        this.getStatus(key, false, { message: (error as Error).message }),
      );
    }
  }
}
```

**NOTE:** The `@InjectQueue` decorator requires BullMQModule to be imported in the HealthModule. Add `BullMQModule` to HealthModule imports.
