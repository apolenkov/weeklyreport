### Auth: dev-login mints real JWT cookies (apps/api/src/modules/auth/auth.routes.ts:224-279)

Endpoint exists ONLY when AUTH_DEV_MODE=true on the API:

```
GET /auth/dev-login?role=lead|head|viewer&email=<seeded-email>&redirect=<path>
  -> sets cookie `customreport_session` (HttpOnly, SameSite=Lax)
  -> redirects to ${frontendUrl}${redirect|defaultByRole}
```

Role defaults: head -> /admin, lead -> /lead/report, viewer -> /reports/history.
When `email` is provided, displayName is looked up from the User table. samlId becomes `dev-${email}`.
buildUserSession then issues a JWT containing { userId, email, name, role, activeReportId, activeReportSectionId }.

Fixtures should call this directly via `request.get(...)` to populate the browser context, NOT navigate the page through it (avoid following the redirect — just keep the Set-Cookie).

### tRPC context resolution (apps/api/src/infrastructure/trpc/trpc.context.ts)

1. If cookie `customreport_session` exists and is valid -> session from JWT.
2. Else if `AUTH_DEV_MODE=true` -> falls back to hardcoded DEV_SESSION (head).
3. Else -> session: null.

For integration tests, the dev-login cookie ALWAYS wins. AUTH_DEV_MODE on the API is required only because dev-login itself is gated by it.

### Web proxy.ts (apps/web/proxy.ts)

Two ways to pass:

- `NEXT_PUBLIC_AUTH_STUB === 'true'` → bypass entirely (used by mock tests).
- Cookie `customreport_session` present → pass.

For integration tests we MUST NOT set NEXT_PUBLIC_AUTH_STUB on the Web server (it would let unauthenticated tests through). Instead, fixtures install the real JWT cookie. Auth-guard tests rely on this — they assert that a fresh context with no cookie redirects to `/?next=...`.

import { execSync } from "node:child_process";
import * as path from "path";

export default async function globalSetup() {
  const dbUrl = process.env.E2E_DATABASE_URL;
  if (!dbUrl) throw new Error("E2E_DATABASE_URL not set");

  const repoRoot = path.resolve(__dirname, "../../../..");

  // Reset and reseed against the e2e database.
  // - migrate reset --force --skip-seed: drops + recreates schema, no auto-seed
  // - then run prisma/seed.ts manually so we control timing and logs
  console.log("[e2e] Resetting integration database...");
  execSync("pnpm exec prisma migrate reset --force --skip-seed", {
    cwd: repoRoot,
    env: { ...process.env, DATABASE_URL: dbUrl },
    stdio: "inherit",
  });

  console.log("[e2e] Seeding integration database...");
  execSync("pnpm exec tsx prisma/seed.ts", {
    cwd: repoRoot,
    env: { ...process.env, DATABASE_URL: dbUrl },
    stdio: "inherit",
  });

  console.log("[e2e] Database ready.");
}

**`apps/web/e2e/integration/fixtures.ts`** — real-API auth via dev-login:

```typescript
import {
  test as base,
  expect,
  type Page,
  type APIRequestContext,
} from "@playwright/test";

const API_URL =
  process.env.NEXT_PUBLIC_API_URL ??
  `http://127.0.0.1:${process.env.E2E_API_PORT ?? 3002}`;

export type Role = "lead" | "head" | "viewer";

/**
 * Mints a real session JWT cookie via the dev-login endpoint and installs it
 * on the given context. Does NOT navigate — caller decides where to go.
 *
 * Why request.get with maxRedirects=0: the endpoint sets a cookie then
 * redirects to a role-default path. We want the cookie, not the redirect.
 */
export async function loginAs(
  request: APIRequestContext,
  page: Page,
  opts: { role: Role; email?: string },
): Promise<void> {
  const url = new URL(`${API_URL}/auth/dev-login`);
  url.searchParams.set("role", opts.role);
  if (opts.email) url.searchParams.set("email", opts.email);

  const res = await request.get(url.toString(), { maxRedirects: 0 });
  if (res.status() !== 302 && res.status() !== 307) {
    throw new Error(`dev-login failed: ${res.status()} ${await res.text()}`);
  }

  // Playwright's APIRequestContext shares cookies with the browser context
  // when both come from the same `context`. We installed it via test.use({}),
  // so the Set-Cookie above is now on `page.context()`. Sanity-check:
  const cookies = await page.context().cookies();
  if (!cookies.find((c) => c.name === "customreport_session")) {
    throw new Error("customreport_session cookie was not set after dev-login");
  }
}

type Fixtures = {
  leadPage: Page; // logged in as Platanov by default
  headPage: Page; // logged in as head
  viewerPage: Page; // logged in as viewer
  unauthedPage: Page; // no cookie — for auth-guard tests
};

export const test = base.extend<Fixtures>({
  leadPage: async ({ browser, request }, use) => {
    const context = await browser.newContext();
    const page = await context.newPage();
    await loginAs(request, page, {
      role: "lead",
      email: "platanov@customreport.local",
    });
    await use(page);
    await context.close();
  },
  headPage: async ({ browser, request }, use) => {
    const context = await browser.newContext();
    const page = await context.newPage();
    await loginAs(request, page, { role: "head" });
    await use(page);
    await context.close();
  },
  viewerPage: async ({ browser, request }, use) => {
    const context = await browser.newContext();
    const page = await context.newPage();
    await loginAs(request, page, { role: "viewer" });
    await use(page);
    await context.close();
  },
  unauthedPage: async ({ browser }, use) => {
    const context = await browser.newContext();
    const page = await context.newPage();
    await use(page);
    await context.close();
  },
});

export { expect };
```

NOTE: Playwright's `request` fixture and `browser.newContext()` do NOT share a cookie jar by default. /**

 * see a known state. Heavier than per-test rollback but simpler.

export async function resetMutations(): Promise<void> {

  // Reset section status to seed state by re-running the relevant seed slices

}
