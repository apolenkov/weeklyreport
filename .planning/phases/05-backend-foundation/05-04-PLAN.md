Create error mapping from domain errors to tRPC error codes:

```typescript
import { TRPCError } from '@trpc/server';
import { ReportSectionDomainError } from '../../modules/report-sections/report-sections.service';
import { ReviewAccessError } from '../../modules/review/review.service';
import { PublicationAccessError } from '../../modules/publication/publication.service';
import { AdminAccessError } from '../../modules/admin/admin.service';
import { PdfExportDomainError } from '../../modules/pdf-export/pdf-export.service';
import { ReminderDomainError } from '../../modules/reminders/reminders.service';
import { CarryOverDomainError } from '../../modules/carry-over/carry-over.service';
import { ArchiveDomainError } from '../../modules/archive/archive.service';

export function mapDomainError(error: unknown): TRPCError {
  if (error instanceof ReportSectionDomainError) {
    return new TRPCError({ code: 'BAD_REQUEST', message: error.message, cause: error });
  }
  if (error instanceof ReviewAccessError || error instanceof AdminAccessError) {
    return new TRPCError({ code: 'FORBIDDEN', message: error.message, cause: error });
  }
  if (error instanceof PublicationAccessError) {
    return new TRPCError({ code: 'CONFLICT', message: error.message, cause: error });
  }
  if (error instanceof PdfExportDomainError || error instanceof ArchiveDomainError) {
    return new TRPCError({ code: 'NOT_FOUND', message: error.message, cause: error });
  }
  if (error instanceof ReminderDomainError || error instanceof CarryOverDomainError) {
    return new TRPCError({ code: 'BAD_REQUEST', message: error.message, cause: error });
  }
  return new TRPCError({ code: 'INTERNAL_SERVER_ERROR', message: 'Unexpected error' });
}
```

```typescript
import { TRPCError } from '@trpc/server';
import { initTRPC } from '@trpc/server';
import type { TrpcContext } from './trpc.context';

const t = initTRPC.context<TrpcContext>().create();

export const publicProcedure = t.procedure;

const isAuthed = t.middleware(({ ctx, next }) => {
  if (!ctx.session) {
    throw new TRPCError({ code: 'UNAUTHORIZED', message: 'Not authenticated' });
  }
  return next({ ctx: { session: ctx.session } });
});

export const protectedProcedure = t.procedure.use(isAuthed);

const isHead = t.middleware(({ ctx, next }) => {
  if (!ctx.session) {
    throw new TRPCError({ code: 'UNAUTHORIZED', message: 'Not authenticated' });
  }
  if (ctx.session.role !== 'head') {
    throw new TRPCError({ code: 'FORBIDDEN', message: 'Head role required' });
  }
  return next({ ctx: { session: ctx.session } });
});

export const headProcedure = t.procedure.use(isHead);

export const router = t.router;
export const mergeRouters = t.mergeRouters;
```

Rewrite the bootstrap function:

```typescript
import 'reflect-metadata';
import { NestFactory } from '@nestjs/core';
import * as trpcExpress from '@trpc/server/adapters/express';
import cookieParser from 'cookie-parser';
import helmet from 'helmet';

import { AppModule } from './app.module';
import { createAppRouter } from './infrastructure/trpc/trpc.router';
import { createContextFactory } from './infrastructure/trpc/trpc.context';
import { validateEnv } from './infrastructure/config/env.schema';
import { createAppLogger } from './lib/logger';

// Import all 8 service classes for app.get() resolution
import { ReportSectionsService } from './modules/report-sections/report-sections.service';
import { ReviewService } from './modules/review/review.service';
import { PublicationService } from './modules/publication/publication.service';
import { PdfExportService } from './modules/pdf-export/pdf-export.service';
import { AdminService } from './modules/admin/admin.service';
import { RemindersService } from './modules/reminders/reminders.service';
import { CarryOverService } from './modules/carry-over/carry-over.service';
import { ArchiveService } from './modules/archive/archive.service';

async function bootstrap() {
  const env = validateEnv();
  const logger = createAppLogger({
    service: 'customreport-api',
    minLevel: env.LOG_LEVEL,
    logDir: env.LOG_DIR,
  });

  const app = await NestFactory.create(AppModule);

  app.use(helmet());
  app.use(cookieParser());
  app.enableCors({ origin: env.FRONTEND_URL, credentials: true });

  // Resolve all 8 services from NestJS DI container
  const appRouter = createAppRouter({
    reportSectionsService: app.get(ReportSectionsService),
    reviewService: app.get(ReviewService),
    publicationService: app.get(PublicationService),
    pdfExportService: app.get(PdfExportService),
    adminService: app.get(AdminService),
    remindersService: app.get(RemindersService),
    carryOverService: app.get(CarryOverService),
    archiveService: app.get(ArchiveService),
  });

  const expressApp = app.getHttpAdapter().getInstance();
  expressApp.use(
    '/trpc',
    trpcExpress.createExpressMiddleware({
      router: appRouter,
      createContext: createContextFactory(app),
    }),
  );

  await app.listen(env.PORT);
  logger.info(`API server listening on port ${env.PORT}`);
}

void bootstrap();
```
