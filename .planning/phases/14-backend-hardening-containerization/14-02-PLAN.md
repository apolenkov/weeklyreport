Modify the runner stage (Stage 4) of Dockerfile.api to reduce image size by copying only production-needed files instead of the entire `/app` directory.

**Current problem:** Line `COPY --from=builder --chown=nodejs:nodejs /app .` copies the entire monorepo including all devDependencies, test files, and source maps.

**Changes to Dockerfile.api Stage 4 (runner):**

Replace the single COPY of entire /app with targeted copies:

```dockerfile
# Stage 4: Production runner
# Use Playwright base image for PDF generation (Chromium pre-installed)
# NOTE: This image is ~1.4GB due to Chromium dependency required for PDF export.
# The 200MB target applies to the web image (alpine-based, ~150MB).
FROM mcr.microsoft.com/playwright:v1.59.0-noble AS runner
RUN groupadd --system --gid 1001 nodejs && \
    useradd --system --uid 1001 nodejs
WORKDIR /app

# Copy pruned monorepo structure with production deps only
COPY --from=builder --chown=nodejs:nodejs /app/package.json ./package.json
COPY --from=builder --chown=nodejs:nodejs /app/pnpm-lock.yaml ./pnpm-lock.yaml
COPY --from=builder --chown=nodejs:nodejs /app/pnpm-workspace.yaml ./pnpm-workspace.yaml
COPY --from=builder --chown=nodejs:nodejs /app/node_modules ./node_modules
COPY --from=builder --chown=nodejs:nodejs /app/apps/api/dist ./apps/api/dist
COPY --from=builder --chown=nodejs:nodejs /app/apps/api/package.json ./apps/api/package.json
COPY --from=builder --chown=nodejs:nodejs /app/apps/api/node_modules ./apps/api/node_modules
COPY --from=builder --chown=nodejs:nodejs /app/packages ./packages

# Prisma client and schema needed at runtime
COPY --from=builder --chown=nodejs:nodejs /app/prisma ./prisma

# Create PDF storage directory
RUN mkdir -p /app/data/pdfs && chown nodejs:nodejs /app/data/pdfs
USER nodejs
WORKDIR /app/apps/api
ENV NODE_ENV=production
EXPOSE 3001
CMD ["node", "dist/main.js"]
```

**Keep stages 1-3 unchanged** -- they already use `turbo prune --docker` correctly.

**Important:** The builder stage may need a `pnpm prune --prod` step added BEFORE the runner copies. Add this as the last line of the builder stage:
```dockerfile
RUN pnpm prune --prod --config.ignore-scripts=true
```
This removes devDependencies from node_modules before they are copied to the runner.

However, test first WITHOUT pnpm prune. If the image builds and runs correctly with targeted copies alone, that is sufficient. The primary size reduction comes from not copying source .ts files, test files, and build tooling.

  <action>
Create a production-specific docker-compose.prod.yml that differs from the dev compose in these ways:

1. **No bind mounts** -- production uses built images, not source mounts
2. **Restart policies** -- `restart: unless-stopped` on all long-running services
3. **Health checks on ALL services** including web (which currently lacks one)
4. **Resource limits** via `deploy.resources.limits`
5. **Logging driver** configuration (json-file with rotation)
6. **No exposed ports for internal services** -- only web (3000) and api (3001) need host ports; postgres and redis use Docker network only

**Exact content for docker-compose.prod.yml:**

```yaml
# Production Docker Compose configuration.
# Usage: docker compose -f docker-compose.prod.yml up -d
#
# Differences from docker-compose.yml:
# - No bind mounts (images are self-contained)
# - Restart policies on all services
# - Health checks on all services including web
# - Resource limits per service
# - Internal-only networking for postgres/redis (no host ports)
# - JSON-file logging with rotation

services:
  postgres:
    image: postgres:15-alpine
    environment:
      POSTGRES_USER: customreport
      POSTGRES_PASSWORD: ${DB_PASSWORD:?DB_PASSWORD must be set}
      POSTGRES_DB: customreport
    volumes:
      - pgdata:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U customreport"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped
    deploy:
      resources:
        limits:
          memory: 512M
    logging:
      driver: json-file
      options:
        max-size: "10m"
        max-file: "3"

  redis:
    image: redis:7-alpine
    command: ["redis-server", "--maxmemory-policy", "noeviction", "--appendonly", "yes"]
    volumes:
      - redisdata:/data
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped
    deploy:
      resources:
        limits:
          memory: 256M
    logging:
      driver: json-file
      options:
        max-size: "10m"
        max-file: "3"

  migrate:
    build:
      context: .
      dockerfile: Dockerfile.api
    command: ["npx", "prisma", "migrate", "deploy"]
    environment:
      DATABASE_URL: postgresql://customreport:${DB_PASSWORD:?DB_PASSWORD must be set}@postgres:5432/customreport
    depends_on:
      postgres:
        condition: service_healthy
    restart: "no"
    logging:
      driver: json-file
      options:
        max-size: "5m"
        max-file: "1"

  api:
    build:
      context: .
      dockerfile: Dockerfile.api
    environment:
      DATABASE_URL: postgresql://customreport:${DB_PASSWORD:?DB_PASSWORD must be set}@postgres:5432/customreport
      REDIS_HOST: redis
      REDIS_PORT: 6379
      NODE_ENV: production
      FRONTEND_URL: ${FRONTEND_URL:-http://localhost:3000}
      AUTH_DEV_MODE: "false"
      AUTH_JWT_SECRET: ${AUTH_JWT_SECRET:?AUTH_JWT_SECRET must be set (min 32 chars)}
      PDF_BASE_URL: http://web:3000
      SMTP_HOST: ${SMTP_HOST:-}
      SMTP_PORT: ${SMTP_PORT:-587}
      SMTP_FROM: ${SMTP_FROM:-noreply@example.com}
    depends_on:
      migrate:
        condition: service_completed_successfully
      redis:
        condition: service_healthy
    healthcheck:
      test: ["CMD-SHELL", "wget -q --spider http://localhost:3001/health/live || exit 1"]
      interval: 15s
      timeout: 5s
      retries: 3
      start_period: 30s
    restart: unless-stopped
    ports:
      - "${API_PORT:-3001}:3001"
    volumes:
      - pdfdata:/app/data/pdfs
    deploy:
      resources:
        limits:
          memory: 2G
    logging:
      driver: json-file
      options:
        max-size: "20m"
        max-file: "5"

  web:
    build:
      context: .
      dockerfile: Dockerfile.web
    environment:
      NEXT_PUBLIC_API_URL: ${NEXT_PUBLIC_API_URL:-http://localhost:3001}
    depends_on:
      api:
        condition: service_healthy
    healthcheck:
      test: ["CMD-SHELL", "wget -q --spider http://localhost:3000/ || exit 1"]
      interval: 15s
      timeout: 5s
      retries: 3
      start_period: 20s
    restart: unless-stopped
    ports:
      - "${WEB_PORT:-3000}:3000"
    deploy:
      resources:
        limits:
          memory: 512M
    logging:
      driver: json-file
      options:
        max-size: "20m"
        max-file: "5"

volumes:
  pgdata:
  redisdata:
  pdfdata:
```

**Key differences from docker-compose.yml:**
- `DB_PASSWORD` is REQUIRED (no default) -- uses `${DB_PASSWORD:?}` syntax
- `AUTH_JWT_SECRET` is REQUIRED -- no insecure default
- `AUTH_DEV_MODE` is `"false"` -- production requires real auth
- postgres/redis have NO `ports:` section -- internal network only
- All services have `restart: unless-stopped`
- All services have `deploy.resources.limits.memory`
- All services have `logging` with json-file rotation
- web has a health check (absent in dev compose)
- Configurable host ports via `API_PORT`, `WEB_PORT` env vars
